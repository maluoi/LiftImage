cmake_minimum_required(VERSION 3.16)
project(LiftImage C)

if(CMAKE_SOURCE_DIR STREQUAL PROJECT_SOURCE_DIR)
	set(LFI_TOP_LEVEL ON)
else()
	set(LFI_TOP_LEVEL OFF)
endif()
option(LFI_ENCODER "Build the encoder into the library"              ON)
option(LFI_TOOLS   "Build the lfi command line tool and the tests" ${LFI_TOP_LEVEL})

# The codec's loops only vectorize when optimized, so it stays optimized in a consumer's Debug build too. These
# flags are scoped to this directory, and MSVC refuses /O2 alongside its Debug runtime checks
string(REGEX REPLACE "/RTC[1csu]+|/Od|/Ob0" "" CMAKE_C_FLAGS_DEBUG "${CMAKE_C_FLAGS_DEBUG}")
function(lfi_optimize target)
	set_target_properties(${target} PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED ON C_EXTENSIONS OFF)
	if(MSVC)
		target_compile_options(${target} PRIVATE /O2)
	else()
		target_compile_options(${target} PRIVATE -O3)
	endif()
endfunction()

set(LFI_SOURCES src/lfi_dec.c src/ent_dec.c)
if(LFI_ENCODER)
	list(APPEND LFI_SOURCES src/lfi_enc.c src/ent_enc.c)
endif()
add_library(lift_image STATIC ${LFI_SOURCES})
add_library(LiftImage::lift_image ALIAS lift_image)
target_include_directories(lift_image PUBLIC src)
set_target_properties(lift_image PROPERTIES POSITION_INDEPENDENT_CODE ON)
lfi_optimize(lift_image)
if(NOT MSVC)
	target_link_libraries(lift_image PUBLIC m)
endif()

if(LFI_TOOLS)
	add_executable(lfi tools/lfi.c tools/stb_impl.c)
	target_include_directories(lfi PRIVATE dist plugins third_party)
	lfi_optimize(lfi)
	if(MSVC)
		target_compile_definitions(lfi PRIVATE _CRT_SECURE_NO_WARNINGS)
	else()
		target_link_libraries(lfi PRIVATE m)
	endif()

	find_package(Python3 COMPONENTS Interpreter)
	if(Python3_FOUND)
		enable_testing()
		add_test(NAME vectors COMMAND Python3::Interpreter ${PROJECT_SOURCE_DIR}/tools/test_vectors.py $<TARGET_FILE:lfi>)
	endif()
endif()
